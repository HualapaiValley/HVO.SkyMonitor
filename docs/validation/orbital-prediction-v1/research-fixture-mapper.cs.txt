    private static OrbitalRecordV1[] Map(JsonElement input, OrbitalElementModel model)
    {
        if (input.ValueKind != JsonValueKind.Array || input.GetArrayLength() > OrbitalContract.MaximumRecords)
            throw new InvalidDataException("Expected bounded GP array.");
        return input.EnumerateArray().Select(row =>
        {
            foreach (var (key, expected) in new[] { ("CENTER_NAME", "EARTH"), ("REF_FRAME", "TEME"), ("TIME_SYSTEM", "UTC"), ("MEAN_ELEMENT_THEORY", "SGP4") })
                if (row.TryGetProperty(key, out var value) && value.GetString() != expected)
                    throw new InvalidDataException($"Unsupported supplied {key}.");
            var catalogValue = row.GetProperty("NORAD_CAT_ID");
            var catalog = catalogValue.ValueKind == JsonValueKind.String ? catalogValue.GetString()! : catalogValue.GetRawText();
            var identity = new OrbitalObjectIdentity("norad", catalog);
            if (!OrbitalContract.IsObjectIdentity(identity)) throw new InvalidDataException("Noncanonical catalog identity.");
            var elements = new OrbitalMeanElements(Number(row, "MEAN_MOTION"), Number(row, "ECCENTRICITY"), Number(row, "INCLINATION"),
                Number(row, "RA_OF_ASC_NODE"), Number(row, "ARG_OF_PERICENTER"), Number(row, "MEAN_ANOMALY"),
                Number(row, "BSTAR"), Number(row, "MEAN_MOTION_DOT"), Number(row, "MEAN_MOTION_DDOT"),
                row.GetProperty("EPHEMERIS_TYPE").GetInt32(), row.GetProperty("CLASSIFICATION_TYPE").GetString()!,
                OptionalLong(row, "ELEMENT_SET_NO"), OptionalLong(row, "REV_AT_EPOCH"));
            return OrbitalContract.CreateRecord(new("", identity, Dataset, null, OptionalText(row, "OBJECT_NAME"),
                OptionalText(row, "OBJECT_ID"), null, null, Utc(row.GetProperty("EPOCH").GetString()!), null, null, model, elements));
        }).ToArray();
    }

    private static object Materialize(string name, OrbitalRecordV1[] records, OrbitalPopulationCoverage coverage,
        string output, OrbitalSnapshotSource source)
    {
        var snapshot = OrbitalSnapshot.Create(records, [source], coverage);
        var manifestBytes = snapshot.SerializeManifest();
        var payloadBytes = snapshot.SerializePayload();
        var parsed = OrbitalSnapshot.Parse(manifestBytes, payloadBytes);
        if (!parsed.IsValid || parsed.Snapshot!.Manifest != snapshot.Manifest && parsed.Snapshot.Manifest.ManifestSha256 != snapshot.Manifest.ManifestSha256)
            throw new InvalidDataException($"Public parse rejected generated canonical input: {parsed.Reason}");
        var directory = Path.Combine(output, name);
        Directory.CreateDirectory(directory);
        var manifestPath = Path.Combine(directory, "manifest.json");
        var payloadPath = Path.Combine(directory, "records.json");
        File.WriteAllBytes(manifestPath, manifestBytes);
        File.WriteAllBytes(payloadPath, payloadBytes);
        return new
        {
            name, manifestIdentitySha256 = snapshot.Manifest.ManifestSha256,
            payloadIdentitySha256 = snapshot.Manifest.NormalizedPayloadSha256,
            snapshot.Manifest.RecordCount, snapshot.Manifest.ObjectCount,
            snapshot.Manifest.MinimumEpochUtc, snapshot.Manifest.MaximumEpochUtc, snapshot.Manifest.Coverage,
            canonicalPublicParseAccepted = parsed.IsValid,
            manifestFile = FileFacts(manifestPath), payloadFile = FileFacts(payloadPath),
            selectedRecords = records.Length <= 6 ? snapshot.Records.Select(r => new { r.ObjectIdentity, r.RecordSha256, r.ObjectName, r.EpochUtc }).ToArray() : null
        };
    }

    private static object SettingsFacts(string path, string output)
    {
        var config = JsonSerializer.Deserialize<CameraModuleConfig>(File.ReadAllBytes(path), Json) ?? throw new InvalidDataException("Missing settings.");
        var rigHash = RigProjectionContextFactory.CreateProfileHashSha256(config.Rig).ToUpperInvariant();
        var readout = config.Rig.Readout is { } r ? SensorReadoutResolver.Resolve(config.Rig.Sensor, r).Geometry :
            new FrameReadoutDescriptor(config.Rig.Sensor.WidthPixels, config.Rig.Sensor.HeightPixels, 0, 0,
                config.Rig.Sensor.WidthPixels, config.Rig.Sensor.HeightPixels, 1, 1, FrameBinningAlgorithm.IdentityV1, 0, 0);
        var readoutHash = CaptureContractJson.ComputeCanonicalJsonSha256(readout);
        var projection = RigProjectionContextFactory.Create(config.Rig);
        var calibration = new AstrometricCalibration(projection, config.Rig.Optics.CalibrationVersion, readoutHash);
        var geometry = new OrbitalGeometryFacts(OrbitalGeometryBasis.ConfiguredRig, rigHash, readoutHash, readout,
            calibration.IdentitySha256.ToUpperInvariant(), null, null, OrbitalProjectionCoordinates.ReadoutView, projection,
            ProjectedSceneImageTransformV1.Identity(projection.WidthPixels, projection.HeightPixels), HorizonPolicy.GeometricHorizon);
        geometry.Validate();
        File.WriteAllBytes(Path.Combine(output, "configured-seattle-geometry.canonical.json"), OrbitalContract.SerializeCanonical(geometry));
        var first = Utc("2026-10-20T13:03:32.287674Z");
        var timing = Enumerable.Range(0, 5).Select(i => new OrbitalTimingFacts(new(first.AddSeconds(5 * i), first.AddSeconds(5 * (i + 1))),
            OrbitalTimingBasis.Estimated, null, OrbitalTimingBasis.Unknown,
            "virtual-poc-logical-shutter-estimated; not sensor-actual", AstrometricClockFacts.NotSupplied)).ToArray();
        foreach (var item in timing) item.Validate();
        File.WriteAllBytes(Path.Combine(output, "estimated-five-exposures.canonical.json"), OrbitalContract.SerializeCanonical(timing));
        return new
        {
            settingsFile = FileFacts(path), observer = config.Observatory, rigHash, readout, canonicalReadoutHash = readoutHash,
            harnessIndentedReadoutHash = Hash(JsonSerializer.SerializeToUtf8Bytes(readout, Json)),
            configuredCalibrationIdentity = calibration.IdentitySha256.ToUpperInvariant(), geometry,
            geometryFile = FileFacts(Path.Combine(output, "configured-seattle-geometry.canonical.json")),
            timingFile = FileFacts(Path.Combine(output, "estimated-five-exposures.canonical.json")),
            limitation = "Configured calibration identity derives only from declared rig projection/readout; no measured calibration or accepted assessment. Pipeline exposure settings remain unchanged; five-second intervals are explicit logical request overrides. Parent tool may choose a different identity convention but must use it consistently."
        };
    }

    private static double Number(JsonElement row, string key) => row.GetProperty(key).GetDouble();
    private static long? OptionalLong(JsonElement row, string key) => row.TryGetProperty(key, out var value) && value.ValueKind != JsonValueKind.Null ? value.GetInt64() : null;
    private static string? OptionalText(JsonElement row, string key) => row.TryGetProperty(key, out var value) && value.ValueKind != JsonValueKind.Null && !string.IsNullOrWhiteSpace(value.GetString()) ? value.GetString() : null;
    private static DateTimeOffset Utc(string text) => DateTimeOffset.Parse(text, CultureInfo.InvariantCulture, DateTimeStyles.AssumeUniversal | DateTimeStyles.AdjustToUniversal);
    private static string Hash(ReadOnlySpan<byte> bytes) => Convert.ToHexString(SHA256.HashData(bytes));
    private static string SourcePath([CallerFilePath] string path = "") => path;
    private static object FileFacts(string path) { var bytes = File.ReadAllBytes(path); return new { path = Path.GetFullPath(path), bytes = bytes.Length, sha256 = Hash(bytes) }; }
    private static object AssemblyFacts(Assembly assembly) => new { file = FileFacts(assembly.Location), name = assembly.GetName().Name, version = assembly.GetName().Version?.ToString(), informationalVersion = assembly.GetCustomAttribute<AssemblyInformationalVersionAttribute>()?.InformationalVersion };
    private static void WriteJson(string path, object value) => File.WriteAllBytes(path, JsonSerializer.SerializeToUtf8Bytes(value, Json));
    private static byte[] ReadPinned(string path, string expectedHash, long expectedLength)
    {
        var bytes = File.ReadAllBytes(path);
        if (bytes.LongLength != expectedLength || Hash(bytes) != expectedHash) throw new InvalidDataException($"Unrecognized retained input: {path}");
        return bytes;
    }
}
